After Match Complete:

public void OnMatchEnd(int placement, int kills)
{
    int coins = CalculateCoins(placement, kills);
    int xp = CalculateXP(placement, kills);
    
    PlayerDataManager.Instance.AddCoins(coins);
    PlayerDataManager.Instance.AddXP(xp);
}

Daily Login Bonus:
------------------
public void OnDailyLogin()
{
    PlayerDataManager.Instance.AddCoins(500);
    ShowNotification("+500 CloudCoins!");
}

Quest Completion:
-----------------
public void OnQuestComplete(Quest quest)
{
    PlayerDataManager.Instance.AddCoins(quest.coinReward);
    PlayerDataManager.Instance.AddXP(quest.xpReward);
}

Battle Pass Tier Unlock:
-------------------------
public void OnTierUnlock(int tier)
{
    var rewards = GetTierRewards(tier);
    PlayerDataManager.Instance.AddCoins(rewards.coins);
}

Character Spawning:
-------------------
public void SpawnPlayer()
{
    var equipped = PlayerDataManager.Instance.GetEquippedCharacter();
    if (equipped != null)
    {
        SpawnCharacterModel(equipped.agentID);
    }
}

================================================================================
TROUBLESHOOTING
================================================================================

"Backend not found":
--------------------
- Check backend is running
- Check URL is http://localhost:8000
- Check no firewall blocking

"Coins show 0 in lobby":
-------------------------
- This is correct for new accounts!
- Check backend JSON has CloudCoins field
- Add test coins to verify sync

"Shop shows wrong amount":
---------------------------
- Check "Use Backend Sync" is enabled
- Check PlayerDataManager in scene
- Restart Unity and login fresh

"Purchase fails":
-----------------
- Check enough coins
- Check character not already owned
- Check backend is running
- Check Console for errors

"Ban not working":
------------------
- Check admin key is correct
- Check player username is exact
- Check BanCheckService in scene
- Check backend is running
