EXAMPLE CODE: FORTNITE TRAIL COMPONENT
================================================================================

Quick Trail Setup Script:
--------------------------

```csharp
using UnityEngine;

namespace TPSBR
{
    public class FortniteRocketTrail : MonoBehaviour
    {
        [Header("Trail")]
        [SerializeField] private TrailRenderer trail;
        [SerializeField] private ParticleSystem smokeTrail;
        
        [Header("Glow")]
        [SerializeField] private Light rocketLight;
        [SerializeField] private float glowIntensity = 2f;
        [SerializeField] private Color glowColor = new Color(1f, 0.6f, 0f); // Orange
        
        private void Start()
        {
            SetupTrail();
            SetupGlow();
        }
        
        private void SetupTrail()
        {
            if (trail == null)
            {
                trail = gameObject.AddComponent<TrailRenderer>();
            }
            
            trail.time = 0.5f;
            trail.minVertexDistance = 0.1f;
            trail.widthCurve = AnimationCurve.Linear(0, 0.3f, 1, 0.05f);
            
            Gradient gradient = new Gradient();
            gradient.SetKeys(
                new GradientColorKey[] {
                    new GradientColorKey(new Color(1f, 0.6f, 0f), 0f),  // Orange
                    new GradientColorKey(new Color(1f, 0.2f, 0f), 1f)   // Red
                },
                new GradientAlphaKey[] {
                    new GradientAlphaKey(1f, 0f),
                    new GradientAlphaKey(0f, 1f)
                }
            );
            trail.colorGradient = gradient;
        }
        
        private void SetupGlow()
        {
            if (rocketLight == null)
            {
                rocketLight = gameObject.AddComponent<Light>();
            }
            
            rocketLight.type = LightType.Point;
            rocketLight.color = glowColor;
            rocketLight.range = 5f;
            rocketLight.intensity = glowIntensity;
        }
    }
}
```

Add this to your rocket projectile prefab!
