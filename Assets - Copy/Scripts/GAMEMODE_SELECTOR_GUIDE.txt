================================================================================
ADVANCED: MODE-SPECIFIC SETTINGS
================================================================================

You can set different settings per gamemode:

private void OnGamemodeButtonClicked()
{
    _currentGamemodeIndex = (_currentGamemodeIndex + 1) % _availableGamemodes.Length;
    _gameplayType = _availableGamemodes[_currentGamemodeIndex];
    
    // Apply gamemode-specific settings
    ApplyGamemodeSettings(_gameplayType);
    
    UpdateGamemodeDisplay();
}

private void ApplyGamemodeSettings(EGameplayType gameplayType)
{
    switch (gameplayType)
    {
        case EGameplayType.BattleRoyale:
            _maxPlayers = 100;
            _searchTimeout = 10f;
            break;
            
        case EGameplayType.TeamDeathmatch:
            _maxPlayers = 20;
            _searchTimeout = 5f;
            break;
            
        case EGameplayType.CaptureTheFlag:
            _maxPlayers = 16;
            _searchTimeout = 8f;
            break;
    }
}
