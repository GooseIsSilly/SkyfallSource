        function banPlayer() {
            const playerID = document.getElementById('banPlayerID').value;
            const reason = document.getElementById('banReason').value;
            const isPermanent = document.getElementById('permanentBan').checked;
            const expiration = document.getElementById('banExpiration').value;

            if (!playerID || !reason) {
                document.getElementById('banError').textContent = 'Please fill all fields';
                return;
            }

            let expiresAt = null;
            if (!isPermanent && expiration) {
                const date = new Date(expiration);
                expiresAt = date.toISOString().slice(0, 19).replace('T', ' ');
            }

            let url = `${API_URL}/admin/ban?AdminKey=${adminKey}&PlayerID=${encodeURIComponent(playerID)}&Reason=${encodeURIComponent(reason)}`;
            if (expiresAt) {
                url += `&ExpiresAt=${encodeURIComponent(expiresAt)}`;
            }

            fetch(url)
                .then(response => response.json())
                .then(data => {
                    if (data.log === 'PlayerBanned' || data.log === 'ok') {
                        closeBanModal();
                        loadBans();
                    } else {
                        document.getElementById('banError').textContent = data.log || 'Error banning player';
                    }
                })
                .catch(error => {
                    document.getElementById('banError').textContent = 'Error connecting to backend';
                    console.error(error);
                });
        }

        function pushUpdate() {
            const version   = document.getElementById('newVersion').value.trim();
            const dlType    = document.querySelector('input[name="dlType"]:checked').value;
            const dlUrl     = document.getElementById('downloadUrl').value.trim();
            const localPath = document.getElementById('localZipPath').value.trim() || 'Skyfall.zip';
            const changelog = document.getElementById('changelog').value.trim();
            const forceUpd  = document.getElementById('forceUpdate').checked;

            if (!version) {
                showUpdateMsg('Please enter a version number.', false);
                return;
            }
            if (dlType === 'url' && !dlUrl) {
                showUpdateMsg('Please enter a download URL.', false);
                return;
            }

            const confirmMsg = forceUpd
                ? `Push v${version} and FORCE-UPDATE all players? Their game will be closed immediately.`
                : `Push v${version}? Players will update next time they open the launcher.`;
            if (!confirm(confirmMsg)) return;

            fetch(`${API_URL}/admin/update/push`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    AdminKey: adminKey,
                    version: version,
                    download_type: dlType,
                    download_url: dlType === 'url' ? dlUrl : '',
                    local_zip_path: dlType === 'local' ? localPath : 'Skyfall.zip',
                    changelog: changelog,
                    force_update: forceUpd,
                })
            })
            .then(r => r.json())
            .then(data => {
                if (data.log === 'ok') {
                    showUpdateMsg(
                        `✓ v${data.version} pushed at ${data.pushed_at}. Launchers will update within 30 seconds.`,
                        true
                    );
                    loadUpdateStatus();
                } else {
                    showUpdateMsg('Error: ' + data.log, false);
                }
            })
            .catch(() => showUpdateMsg('Connection error', false));
        }

                .catch(() => showShopMsg('shopActionMsg', 'Connection error', false));
        }

        function forceRotateShop(section) {
            if (!confirm(`Force rotate ${section} shop?`)) return;
            fetch(`${API_URL}/admin/shop/forcerotate?AdminKey=${adminKey}&section=${section}`)
                .then(r => r.json())
                .then(data => {
                    if (data.log === 'ok') {
                        showShopMsg('shopActionMsg', `Forced rotation for ${section}.`, true);
                        loadShopQueue();
                        loadShopCatalog();
                    } else {
                        showShopMsg('shopActionMsg', 'Error: ' + data.log, false);
                    }
                })
                .catch(() => showShopMsg('shopActionMsg', 'Connection error', false));
        }

        function loadNews() {
            fetch(`${API_URL}/admin/news/get?AdminKey=${adminKey}`)
                .then(r => r.json())
                .then(data => {
                    if (data.log !== 'ok') return;
                    const items = data.news;
                    document.getElementById('totalNews').textContent = items.length;
                    document.getElementById('publishedNews').textContent = items.filter(n => n.published).length;

                    const tbody = document.getElementById('newsBody');
                    tbody.innerHTML = '';
                    items.forEach(n => {
                        const row = tbody.insertRow();
                        const tagColors = { NEW: '#16a085', UPDATE: '#2980b9', EVENT: '#8e44ad' };
                        const color = tagColors[n.tag] || '#555';
                        row.innerHTML = `
                            <td><span class="badge" style="background:${color}; color:white;">${n.tag}</span></td>
                            <td><strong>${n.title}</strong></td>
                            <td style="max-width:300px; word-break:break-word; font-size:13px;">${n.body}</td>
                            <td style="white-space:nowrap;">${n.date}</td>
                            <td>${n.published
                                ? '<span class="badge actioned">Published</span>'
                                : '<span class="badge pending">Hidden</span>'}</td>
                            <td style="white-space:nowrap;">
                                <button onclick="editNews(${JSON.stringify(n).replace(/"/g, '&quot;')})">Edit</button>
                                <button class="danger" onclick="deleteNews('${n.id}')">Delete</button>
                            </td>
                        `;
                    });
                })
                .catch(err => console.error('Error loading news:', err));
        }

        function showNewsModal(item) {
            document.getElementById('newsModalTitle').textContent = item ? 'Edit Post' : 'New Post';
            document.getElementById('newsEditId').value = item ? item.id : '';
            document.getElementById('newsTag').value = item ? item.tag : 'NEW';
            document.getElementById('newsTitle').value = item ? item.title : '';
            document.getElementById('newsBody').value = item ? item.body : '';
            document.getElementById('newsDate').value = item ? item.date : '';
            document.getElementById('newsPublished').checked = item ? item.published : true;
            document.getElementById('newsModalMsg').textContent = '';
            document.getElementById('newsModal').classList.add('active');
        }

        function editNews(item) {
            showNewsModal(item);
        }

        function closeNewsModal() {
            document.getElementById('newsModal').classList.remove('active');
        }

        function saveNews() {
            const id        = document.getElementById('newsEditId').value;
            const tag       = document.getElementById('newsTag').value;
            const title     = document.getElementById('newsTitle').value.trim();
            const body      = document.getElementById('newsBody').value.trim();
            const date      = document.getElementById('newsDate').value.trim();
            const published = document.getElementById('newsPublished').checked;

            if (!title || !body) {
                document.getElementById('newsModalMsg').textContent = 'Title and body are required.';
                document.getElementById('newsModalMsg').style.color = '#e74c3c';
                return;
            }

            const isEdit = !!id;
            const url    = isEdit ? `${API_URL}/admin/news/edit` : `${API_URL}/admin/news/add`;
            const payload = isEdit
                ? { AdminKey: adminKey, id, tag, title, body, date, published }
                : { AdminKey: adminKey, tag, title, body, date, published };

            fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            })
            .then(r => r.json())
            .then(data => {
                if (data.log === 'ok') {
                    closeNewsModal();
                    loadNews();
                } else {
                    document.getElementById('newsModalMsg').textContent = 'Error: ' + data.log;
                    document.getElementById('newsModalMsg').style.color = '#e74c3c';
                }
            })
            .catch(() => {
                document.getElementById('newsModalMsg').textContent = 'Connection error';
                document.getElementById('newsModalMsg').style.color = '#e74c3c';
            });
        }

        function deleteNews(id) {
            if (!confirm('Delete this news post? This cannot be undone.')) return;
            fetch(`${API_URL}/admin/news/delete?AdminKey=${adminKey}&id=${id}`)
                .then(r => r.json())
                .then(data => {
                    if (data.log === 'ok') {
                        loadNews();
                    } else {
                        alert('Error: ' + data.log);
                    }
                })
                .catch(() => alert('Connection error'));
        }
