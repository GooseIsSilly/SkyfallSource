LIVE_EVENT_FILE = "live_event.json"


def _load_live_event() -> dict:
    """Loads the live event state from disk, returning defaults if missing."""
    if os.path.exists(LIVE_EVENT_FILE):
        with open(LIVE_EVENT_FILE, "r") as f:
            return json.load(f)
    return {
        "EventStartTimeEST": "",
        "ForceTriggered": False,
        "ForceTriggerTimestamp": "",
        "SeasonEndedEnabled": False,
        "LastUpdated": "",
    }


def _save_live_event(data: dict):
    data["LastUpdated"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(LIVE_EVENT_FILE, "w") as f:
        json.dump(data, f, indent=4)

def readuserfile(ID):
    with open("Accounts/" + ID + ".json") as file:

def addtoken(ID):
    Token = "".join(
        random.choice(string.ascii_uppercase + string.digits) for _ in range(30)
    )
    TokenDict[ID] = Token
    _save_tokens()
    print("Created token for", ID)
    return Token


def removetoken(ID):
    if ID in TokenDict:
        del TokenDict[ID]
        _save_tokens()
        print(ID + "'s token deleted.")
        return
    else:
        print(ID + "'s token could not be deleted as it didn't exist")
        return


def getuserfilebytoken(Token):
    FoundPlayer = [key for key, value in TokenDict.items() if value == Token]
    if len(FoundPlayer) == 1:
        return readuserfile(FoundPlayer[0])
    else:
        print("Token not found")
        return "MissingToken"

def createaccount(ID: str, Pass: str):
    if not os.path.exists("Accounts/" + ID + ".json"):
        if len(ID) <= 2:
            print("Account was not created as ID is too short")
            return {"log": "IDTooShort"}
        if len(Pass) <= 8:
            print("Account was not created as pass is too short")
            return {"log": "PassTooShort"}
        Hash = bcrypt.hashpw(Pass.encode("utf-8"), bcrypt.gensalt())
        createuserfile(ID)
        setpass(ID, Hash)
        token = addtoken(ID)
        return {"log": "ok", "token": token, "Token": token}
    else:
        print(
            "Could not create account", ID, "as another account already has that name"
        )
        return {"log": "AccountAlreadyExists"}


def auth(ID: str, Pass: str):
    if not os.path.exists("Accounts/" + ID + ".json"):
        print("Connection failed, username does not exist.")
        return {"log": "UserNameNotExist"}

    # Check ban status before authentication
    ban_status = isbanned(ID)
    if ban_status["banned"]:
        print(f"{ID} attempted to login but is banned.")
        return {
            "log": "AccountBanned",
            "reason": ban_status["reason"],
            "expiresAt": ban_status.get("expiresAt"),
        }

    JFile = readuserfile(ID)
    AccPass = JFile["Pass"]
    if bcrypt.checkpw(Pass.encode("utf-8"), AccPass.encode("utf-8")):
        print(ID, "connected successfully.")
        token = addtoken(ID)
        return {"log": "ok", "token": token, "Token": token}

    else:
        print(ID, "input a wrong pass.")
        return {"log": "PassWrong"}

def banplayer(ID: str, Reason: str, BannedBy: str, ExpiresAt: str = None):
    if not os.path.exists("Accounts/" + ID + ".json"):
        return {"log": "PlayerNotFound"}

    JFile = readuserfile(ID)
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    JFile["BanStatus"] = {
        "IsBanned": True,
        "BanReason": Reason,
        "BannedAt": current_time,
        "BannedBy": BannedBy,
        "ExpiresAt": ExpiresAt,
    }

    with open("Accounts/" + ID + ".json", "w") as savefile:
        json.dump(JFile, savefile, indent=4)

    if not os.path.exists("Bans"):
        os.makedirs("Bans")

    ban_record = {
        "PlayerID": ID,
        "Reason": Reason,
        "BannedBy": BannedBy,
        "BannedAt": current_time,
        "ExpiresAt": ExpiresAt,
        "IsActive": True,
    }

    with open(f"Bans/{ID}.json", "w") as banfile:
        json.dump(ban_record, banfile, indent=4)

    removetoken(ID)
    print(f"{ID} has been banned. Reason: {Reason}")
    return {"log": "PlayerBanned"}


def unbanplayer(ID: str):
    if not os.path.exists("Accounts/" + ID + ".json"):
        return {"log": "PlayerNotFound"}

    JFile = readuserfile(ID)
    JFile["BanStatus"]["IsBanned"] = False

    with open("Accounts/" + ID + ".json", "w") as savefile:
        json.dump(JFile, savefile, indent=4)

    if os.path.exists(f"Bans/{ID}.json"):
        if not os.path.exists("Bans/Archive"):
            os.makedirs("Bans/Archive")
        shutil.move(f"Bans/{ID}.json", f"Bans/Archive/{ID}.json")

    print(f"{ID} has been unbanned.")
    return {"log": "PlayerUnbanned"}


def checkbanexpired(ID: str):
    JFile = readuserfile(ID)

    if "BanStatus" not in JFile or not JFile["BanStatus"]["IsBanned"]:
        return False

    expires_at = JFile["BanStatus"].get("ExpiresAt")
    if expires_at is None:
        return False

    try:
        expire_time = datetime.strptime(expires_at, "%Y-%m-%d %H:%M:%S")
        if datetime.now() >= expire_time:
            unbanplayer(ID)
            return True
    except:
        pass

    return False


def isbanned(ID: str):
    if not os.path.exists("Accounts/" + ID + ".json"):
        return {"banned": False}

    checkbanexpired(ID)
    JFile = readuserfile(ID)

    if "BanStatus" not in JFile:
        JFile["BanStatus"] = {
            "IsBanned": False,
            "BanReason": "",
            "BannedAt": "",
            "BannedBy": "",
            "ExpiresAt": None,
        }
        with open("Accounts/" + ID + ".json", "w") as savefile:
            json.dump(JFile, savefile, indent=4)
        return {"banned": False}

    if JFile["BanStatus"]["IsBanned"]:
        return {
            "banned": True,
            "reason": JFile["BanStatus"]["BanReason"],
            "bannedAt": JFile["BanStatus"]["BannedAt"],
            "expiresAt": JFile["BanStatus"]["ExpiresAt"],
        }

    return {"banned": False}


def getactivebans():
    if not os.path.exists("Bans"):
        return []

    ban_files = glob.glob("Bans/*.json")
    active_bans = []

    for ban_file in ban_files:
        with open(ban_file, "r") as f:
            ban_data = json.load(f)
            if ban_data.get("IsActive", True):
                active_bans.append(ban_data)

    return active_bans

def canreport(ID: str):
    JFile = readuserfile(ID)

    if "ReportData" not in JFile:
        JFile["ReportData"] = {"LastReportTime": "", "ReportCount": 0}
        with open("Accounts/" + ID + ".json", "w") as savefile:
            json.dump(JFile, savefile, indent=4)
        return True

    last_report = JFile["ReportData"].get("LastReportTime", "")
    if last_report == "":
        return True

    try:
        last_time = datetime.strptime(last_report, "%Y-%m-%d %H:%M:%S")
        time_diff = (datetime.now() - last_time).total_seconds()

        if time_diff < 3600:
            report_count = JFile["ReportData"].get("ReportCount", 0)
            if report_count >= 5:
                return False
        else:
            JFile["ReportData"]["ReportCount"] = 0
            with open("Accounts/" + ID + ".json", "w") as savefile:
                json.dump(JFile, savefile, indent=4)
    except:
        pass

    return True


def updatereportcount(ID: str):
    JFile = readuserfile(ID)
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    if "ReportData" not in JFile:
        JFile["ReportData"] = {"LastReportTime": current_time, "ReportCount": 1}
    else:
        last_report = JFile["ReportData"].get("LastReportTime", "")
        try:
            last_time = datetime.strptime(last_report, "%Y-%m-%d %H:%M:%S")
            time_diff = (datetime.now() - last_time).total_seconds()

            if time_diff >= 3600:
                JFile["ReportData"]["ReportCount"] = 1
            else:
                JFile["ReportData"]["ReportCount"] = (
                    JFile["ReportData"].get("ReportCount", 0) + 1
                )
        except:
            JFile["ReportData"]["ReportCount"] = 1

        JFile["ReportData"]["LastReportTime"] = current_time

    with open("Accounts/" + ID + ".json", "w") as savefile:
        json.dump(JFile, savefile, indent=4)

# Parties are session-based and stored in memory only.
# PartyDict structure:
#   key   = party_code (6-char uppercase string)
#   value = {
#       "leader":  str,        # AccountName of the leader
#       "members": list[str],  # AccountNames in the party
#       "mode":    str         # "solo" | "duo" | "squad"
#   }

PartyDict = {}  # party_code -> party data
PlayerPartyMap = {}  # account_name -> party_code

PARTY_MAX_SIZES = {"solo": 4, "duo": 4, "trio": 3, "squad": 4}


def generate_party_code():
    """Generates a unique 6-character alphanumeric party code."""
    while True:
        code = "".join(random.choices(string.ascii_uppercase + string.digits, k=6))
        if code not in PartyDict:
            return code


def get_party_for_player(account_name: str):
    """Returns the party dict for a player, or None if they are not in one."""
    code = PlayerPartyMap.get(account_name)
    if code:
        return PartyDict.get(code)
    return None


def remove_player_from_party(account_name: str):
    """
    Removes a player from their current party.
    Disbands the party if it becomes empty.
    If the leader leaves, the next member is promoted.
    """
    code = PlayerPartyMap.pop(account_name, None)
    if not code or code not in PartyDict:
        return

    party = PartyDict[code]
    if account_name in party["members"]:
        party["members"].remove(account_name)

    if len(party["members"]) == 0:
        del PartyDict[code]
        return

    # Promote a new leader if the current one left
    if party["leader"] == account_name:
        party["leader"] = party["members"][0]


@app.get("/player/lookup")
def appplayer_lookup(Username: str):
    """
    Checks whether an account with the given username exists.
    The lookup is case-insensitive — the resolved AccountName with
    its original casing is returned on success.
    No authentication required — used for friend-adding by username.
    """
    if not Username or len(Username) < 3:
        return {"log": "UsernameTooShort"}

    # Try exact match first
    if os.path.exists("Accounts/" + Username + ".json"):
        data = readuserfile(Username)
        ban_status = isbanned(data["AccountName"])
        if ban_status["banned"]:
            return {"log": "PlayerNotFound"}
        return {"log": "ok", "username": data["AccountName"]}

    # Fall back to case-insensitive scan of the Accounts directory
    if os.path.exists("Accounts"):
        for filename in os.listdir("Accounts"):
            if filename.lower() == Username.lower() + ".json":
                resolved_id = filename[:-5]  # strip .json
                data = readuserfile(resolved_id)
                ban_status = isbanned(data["AccountName"])
                if ban_status["banned"]:
                    return {"log": "PlayerNotFound"}
                return {"log": "ok", "username": data["AccountName"]}

    return {"log": "PlayerNotFound"}


@app.get("/party/create")
def appcreate_party(Token: str, Mode: str = "duo"):
    JFile = getuserfilebytoken(Token)
    if JFile == "MissingToken":
        return {"log": "TokenDoesNotExist"}

    account_name = JFile["AccountName"]
    mode = Mode.lower()

    if mode not in PARTY_MAX_SIZES:
        return {"log": "InvalidMode"}

    # Leave any existing party first
    remove_player_from_party(account_name)

    code = generate_party_code()
    PartyDict[code] = {
        "leader": account_name,
        "members": [account_name],
        "mode": mode,
        "fill": True,  # Default to fill
        "ready": {},   # member_name -> bool ready state
    }
    PlayerPartyMap[account_name] = code

    print(f"[Party] {account_name} created party {code} ({mode})")
    return {
        "log": "ok",
        "partyCode": code,
        "leader": account_name,
        "members": ",".join(PartyDict[code]["members"]),
        "mode": mode,
        "fill": True,
    }

    JFile = getuserfilebytoken(Token)

    code = PartyCode.upper()

    if code not in PartyDict:

    party = PartyDict[code]
    max_size = PARTY_MAX_SIZES.get(party["mode"], 1)

    if account_name in party["members"]:
        return {"log": "AlreadyInParty"}

    if len(party["members"]) >= max_size:
        return {"log": "PartyFull"}

    remove_player_from_party(account_name)

    party["members"].append(account_name)
    PlayerPartyMap[account_name] = code

    print(f"[Party] {account_name} joined party {code}")
    return {
        "log": "ok",
        "leader": party["leader"],
        "members": ",".join(party["members"]),
        "mode": party["mode"],
    }

# ============================================================================
# FRIEND REQUEST SYSTEM
# ============================================================================
# Stored persistently in each account's JSON file under the "Friends" key.
#
# Account "Friends" structure:
#   {
#       "list":     [ "AccountName", ... ],          # confirmed friends
#       "incoming": [ "AccountName", ... ],          # pending requests received
#       "outgoing": [ "AccountName", ... ]           # pending requests sent
#   }
# ============================================================================

def _ensure_friends(jfile: dict) -> dict:
    """Guarantees the Friends sub-dict exists on an account file."""
    if "Friends" not in jfile:
        jfile["Friends"] = {"list": [], "incoming": [], "outgoing": []}
    else:
        jfile["Friends"].setdefault("list", [])
        jfile["Friends"].setdefault("incoming", [])
        jfile["Friends"].setdefault("outgoing", [])
    return jfile


def _save_userfile(account_name: str, jfile: dict):
    with open("Accounts/" + account_name + ".json", "w") as f:
        json.dump(jfile, f, indent=4)


@app.get("/friends/list")

    jfile = getuserfilebytoken(Token)
    if jfile == "MissingToken":
        return {"log": "TokenDoesNotExist"}

    jfile = _ensure_friends(jfile)
    friends = jfile["Friends"]
    return {
        "log": "ok",
        "friends":  friends["list"],
        "incoming": friends["incoming"],
        "outgoing": friends["outgoing"],
    }


@app.get("/friends/request/send")
def appfriends_request_send(Token: str, TargetID: str):
    """Sends a friend request from the caller to TargetID."""
    jfile = getuserfilebytoken(Token)
    if jfile == "MissingToken":
        return {"log": "TokenDoesNotExist"}

    sender = jfile["AccountName"]

    if not os.path.exists("Accounts/" + TargetID + ".json"):
        # Try case-insensitive lookup
        resolved = None
        if os.path.exists("Accounts"):
            for filename in os.listdir("Accounts"):
                if filename.lower() == TargetID.lower() + ".json":
                    resolved = filename[:-5]
                    break
        if resolved is None:
            return {"log": "PlayerNotFound"}
        TargetID = resolved

    if sender == TargetID:
        return {"log": "CannotAddSelf"}

    jfile = _ensure_friends(jfile)

    if TargetID in jfile["Friends"]["list"]:
        return {"log": "AlreadyFriends"}

    if TargetID in jfile["Friends"]["outgoing"]:
        return {"log": "RequestAlreadySent"}

    # If the target already sent us a request, auto-accept
    if TargetID in jfile["Friends"]["incoming"]:
        jfile["Friends"]["incoming"].remove(TargetID)
        if TargetID not in jfile["Friends"]["list"]:
            jfile["Friends"]["list"].append(TargetID)
        _save_userfile(sender, jfile)

        target_jfile = readuserfile(TargetID)
        target_jfile = _ensure_friends(target_jfile)
        if sender in target_jfile["Friends"]["outgoing"]:
            target_jfile["Friends"]["outgoing"].remove(sender)
        if sender not in target_jfile["Friends"]["list"]:
            target_jfile["Friends"]["list"].append(sender)
        _save_userfile(TargetID, target_jfile)

        print(f"[Friends] Auto-accepted mutual request: {sender} <-> {TargetID}")
        return {"log": "AutoAccepted", "friend": TargetID}

    # Record outgoing on sender's side
    if TargetID not in jfile["Friends"]["outgoing"]:
        jfile["Friends"]["outgoing"].append(TargetID)
    _save_userfile(sender, jfile)

    # Record incoming on target's side
    target_jfile = readuserfile(TargetID)
    target_jfile = _ensure_friends(target_jfile)
    if sender not in target_jfile["Friends"]["incoming"]:
        target_jfile["Friends"]["incoming"].append(sender)
    _save_userfile(TargetID, target_jfile)

    print(f"[Friends] {sender} sent friend request to {TargetID}")
    return {"log": "ok", "target": TargetID}


@app.get("/friends/request/accept")
def appfriends_request_accept(Token: str, FromID: str):
    """Accepts a pending incoming friend request from FromID."""
    jfile = getuserfilebytoken(Token)
    if jfile == "MissingToken":
        return {"log": "TokenDoesNotExist"}

    accepter = jfile["AccountName"]
    jfile = _ensure_friends(jfile)

    if FromID not in jfile["Friends"]["incoming"]:
        return {"log": "NoSuchRequest"}

    jfile["Friends"]["incoming"].remove(FromID)
    if FromID not in jfile["Friends"]["list"]:
        jfile["Friends"]["list"].append(FromID)
    _save_userfile(accepter, jfile)

    # Mirror on the sender's side
    if os.path.exists("Accounts/" + FromID + ".json"):
        from_jfile = readuserfile(FromID)
        from_jfile = _ensure_friends(from_jfile)
        if accepter in from_jfile["Friends"]["outgoing"]:
            from_jfile["Friends"]["outgoing"].remove(accepter)
        if accepter not in from_jfile["Friends"]["list"]:
            from_jfile["Friends"]["list"].append(accepter)
        _save_userfile(FromID, from_jfile)

    print(f"[Friends] {accepter} accepted friend request from {FromID}")
    return {"log": "ok", "friend": FromID}


@app.get("/friends/request/decline")
def appfriends_request_decline(Token: str, FromID: str):
    """Declines or cancels a friend request."""
    jfile = getuserfilebytoken(Token)
    if jfile == "MissingToken":
        return {"log": "TokenDoesNotExist"}

    decliner = jfile["AccountName"]
    jfile = _ensure_friends(jfile)

    # Remove from incoming (if declining a received request)
    if FromID in jfile["Friends"]["incoming"]:
        jfile["Friends"]["incoming"].remove(FromID)
    # Remove from outgoing (if cancelling a sent request)
    if FromID in jfile["Friends"]["outgoing"]:
        jfile["Friends"]["outgoing"].remove(FromID)
    _save_userfile(decliner, jfile)

    # Clean up the other side's outgoing / incoming
    if os.path.exists("Accounts/" + FromID + ".json"):
        from_jfile = readuserfile(FromID)
        from_jfile = _ensure_friends(from_jfile)
        if decliner in from_jfile["Friends"]["outgoing"]:
            from_jfile["Friends"]["outgoing"].remove(decliner)
        if decliner in from_jfile["Friends"]["incoming"]:
            from_jfile["Friends"]["incoming"].remove(decliner)
        _save_userfile(FromID, from_jfile)

    print(f"[Friends] {decliner} declined/cancelled request with {FromID}")
    return {"log": "ok"}

# ============================================================================
# ITEM SHOP SYSTEM
# ============================================================================
#
# Files used:
#   shop_catalog.json  — current live daily/weekly items + reset timestamps
#   shop_queue.json    — admin-managed queue of upcoming item IDs
#
# Rotation schedule:
#   Daily  items rotate every 24 hours (6 items drawn from the queue or random pool)
#   Weekly items rotate every  7 days  (4 items drawn from the queue or random pool)
#
# The rotation is checked lazily on every /shop/catalog call so no background
# scheduler process is required.
# ============================================================================

SHOP_CATALOG_FILE = "shop_catalog.json"
SHOP_QUEUE_FILE   = "shop_queue.json"

# How many items appear in each section
DAILY_SLOT_COUNT  = 6
WEEKLY_SLOT_COUNT = 4

# Fallback item pool used when the queue is empty.
# Replace / extend these IDs to match your ShopDatabase asset IDs.
FALLBACK_ITEM_POOL = [
    "agent_default",
    "agent_shadow",
    "agent_arctic",
    "agent_inferno",
    "agent_storm",
    "agent_ghost",
    "agent_neon",
    "agent_crimson",
    "agent_cobalt",
    "agent_emerald",
]


def _load_shop_catalog() -> dict:
    """Returns the current shop catalog from disk, or an empty default."""
    if os.path.exists(SHOP_CATALOG_FILE):
        with open(SHOP_CATALOG_FILE, "r") as f:
            return json.load(f)
    return {
        "daily":        [],
        "weekly":       [],
        "daily_reset":  "",
        "weekly_reset": "",
        "last_updated": "",
    }


def _save_shop_catalog(data: dict):
    data["last_updated"] = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
    with open(SHOP_CATALOG_FILE, "w") as f:
        json.dump(data, f, indent=4)


def _load_shop_queue() -> dict:
    """Returns the admin-managed item queue from disk."""
    if os.path.exists(SHOP_QUEUE_FILE):
        with open(SHOP_QUEUE_FILE, "r") as f:
            return json.load(f)
    return {"daily_queue": [], "weekly_queue": []}


def _save_shop_queue(data: dict):
    with open(SHOP_QUEUE_FILE, "w") as f:
        json.dump(data, f, indent=4)


def _pick_items(queue: list, count: int) -> list:
    """
    Draws `count` item IDs from the front of `queue`.
    Fills any remaining slots from FALLBACK_ITEM_POOL (deduplicated, random order).
    Mutates `queue` in place by consuming items that were used.
    """
    picked = []

    # Pull from the queue first
    while queue and len(picked) < count:
        picked.append(queue.pop(0))

    # Fill remaining slots from the fallback pool (avoid duplicates)
    available = [i for i in FALLBACK_ITEM_POOL if i not in picked]
    random.shuffle(available)
    picked.extend(available[: count - len(picked)])

    return picked


def _build_item_entry(item_id: str) -> dict:
    """
    Builds a single shop item dict.
    Rarity and price are looked up from a lightweight in-memory table.
    Extend ITEM_META to match your actual ShopDatabase entries.
    """
    ITEM_META = {
        "agent_default": {"price": 0,    "rarity": "common",    "type": "agent"},
        "agent_shadow":  {"price": 800,  "rarity": "rare",      "type": "agent"},
        "agent_arctic":  {"price": 800,  "rarity": "rare",      "type": "agent"},
        "agent_inferno": {"price": 1200, "rarity": "epic",      "type": "agent"},
        "agent_storm":   {"price": 1200, "rarity": "epic",      "type": "agent"},
        "agent_ghost":   {"price": 1500, "rarity": "epic",      "type": "agent"},
        "agent_neon":    {"price": 2000, "rarity": "legendary", "type": "agent"},
        "agent_crimson": {"price": 2000, "rarity": "legendary", "type": "agent"},
        "agent_cobalt":  {"price": 500,  "rarity": "uncommon",  "type": "agent"},
        "agent_emerald": {"price": 500,  "rarity": "uncommon",  "type": "agent"},
    }
    meta = ITEM_META.get(item_id, {"price": 800, "rarity": "rare", "type": "agent"})
    return {"id": item_id, **meta}


def _needs_daily_rotation(catalog: dict) -> bool:
    if not catalog.get("daily_reset"):
        return True
    try:
        reset_time = datetime.strptime(catalog["daily_reset"], "%Y-%m-%dT%H:%M:%SZ")
        return datetime.utcnow() >= reset_time
    except ValueError:
        return True


def _needs_weekly_rotation(catalog: dict) -> bool:
    if not catalog.get("weekly_reset"):
        return True
    try:
        reset_time = datetime.strptime(catalog["weekly_reset"], "%Y-%m-%dT%H:%M:%SZ")
        return datetime.utcnow() >= reset_time
    except ValueError:
        return True


def _rotate_shop_if_needed(catalog: dict, queue: dict) -> bool:
    """
    Checks whether daily/weekly rotations are due and applies them.
    Returns True if any rotation occurred (caller should save both files).
    """
    from datetime import timedelta

    changed = False

    if _needs_daily_rotation(catalog):
        new_ids = _pick_items(queue["daily_queue"], DAILY_SLOT_COUNT)
        catalog["daily"]       = [_build_item_entry(i) for i in new_ids]
        catalog["daily_reset"] = (
            datetime.utcnow() + timedelta(hours=24)
        ).strftime("%Y-%m-%dT%H:%M:%SZ")
        print(f"[Shop] Daily rotation applied: {new_ids}")
        changed = True

    if _needs_weekly_rotation(catalog):
        new_ids = _pick_items(queue["weekly_queue"], WEEKLY_SLOT_COUNT)
        catalog["weekly"]       = [_build_item_entry(i) for i in new_ids]
        catalog["weekly_reset"] = (
            datetime.utcnow() + timedelta(days=7)
        ).strftime("%Y-%m-%dT%H:%M:%SZ")
        print(f"[Shop] Weekly rotation applied: {new_ids}")
        changed = True

    return changed

@app.get("/shop/catalog")
def get_shop_catalog():
    """
    Returns the current daily and weekly shop items.
    Automatically rotates items when their reset timer has expired.
    Unity calls this on shop open via BackendServiceManager.GetShopCatalog().
    """
    catalog = _load_shop_catalog()
    queue   = _load_shop_queue()

    if _rotate_shop_if_needed(catalog, queue):
        _save_shop_catalog(catalog)
        _save_shop_queue(queue)

    return {
        "log":          "ok",
        "daily":        catalog["daily"],
        "weekly":       catalog["weekly"],
        "daily_reset":  catalog["daily_reset"],
        "weekly_reset": catalog["weekly_reset"],
    }

class ShopQueueAddRequest(BaseModel):
    AdminKey: str
    section: str   # "daily" or "weekly"
    item_ids: list[str]


@app.post("/admin/shop/queue/add")
def admin_shop_queue_add(req: ShopQueueAddRequest):
    """
    Appends one or more item IDs to the daily or weekly rotation queue.
    Items are served FIFO — the first item added will be the first used on the
    next rotation.

    Body example:
        { "AdminKey": "...", "section": "daily", "item_ids": ["agent_neon", "agent_storm"] }
    """
    if req.AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    if req.section not in ("daily", "weekly"):
        return {"log": "InvalidSection", "expected": "daily or weekly"}

    queue = _load_shop_queue()
    key   = f"{req.section}_queue"
    queue[key].extend(req.item_ids)
    _save_shop_queue(queue)

    print(f"[Shop] Admin queued {req.item_ids} for {req.section}")
    return {"log": "ok", "section": req.section, "queued": req.item_ids, "queue_length": len(queue[key])}


@app.get("/admin/shop/queue/get")
def admin_shop_queue_get(AdminKey: str):
    """Returns the full pending queue for both daily and weekly sections."""
    if AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    queue = _load_shop_queue()
    return {"log": "ok", "queue": queue}


class ShopQueueClearRequest(BaseModel):
    AdminKey: str
    section: str   # "daily", "weekly", or "all"


@app.post("/admin/shop/queue/clear")
def admin_shop_queue_clear(req: ShopQueueClearRequest):
    """Clears the pending queue for the given section ('daily', 'weekly', or 'all')."""
    if req.AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    if req.section not in ("daily", "weekly", "all"):
        return {"log": "InvalidSection", "expected": "daily, weekly, or all"}

    queue = _load_shop_queue()
    if req.section in ("daily", "all"):
        queue["daily_queue"] = []
    if req.section in ("weekly", "all"):
        queue["weekly_queue"] = []
    _save_shop_queue(queue)

    return {"log": "ok", "cleared": req.section}


@app.get("/admin/shop/forcerotate")
def admin_shop_force_rotate(AdminKey: str, section: str = "all"):
    """
    Immediately triggers a rotation for 'daily', 'weekly', or 'all',
    regardless of whether the reset timer has expired.
    """
    if AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    if section not in ("daily", "weekly", "all"):
        return {"log": "InvalidSection", "expected": "daily, weekly, or all"}

    from datetime import timedelta

    catalog = _load_shop_catalog()
    queue   = _load_shop_queue()

    rotated = []

    if section in ("daily", "all"):
        # Force expiry so _rotate_shop_if_needed picks it up
        catalog["daily_reset"] = ""
    if section in ("weekly", "all"):
        catalog["weekly_reset"] = ""

    if _rotate_shop_if_needed(catalog, queue):
        _save_shop_catalog(catalog)
        _save_shop_queue(queue)
        rotated.append(section)

    return {"log": "ok", "rotated": rotated, "catalog": {"daily": catalog["daily"], "weekly": catalog["weekly"]}}


@app.get("/admin/shop/catalog/get")
def admin_shop_catalog_get(AdminKey: str):
    """Returns the full live catalog including reset timestamps (admin view)."""
    if AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    catalog = _load_shop_catalog()
    return {"log": "ok", "catalog": catalog}

NEWS_FILE = "news.json"


def _load_news() -> list:
    if os.path.exists(NEWS_FILE):
        with open(NEWS_FILE, "r") as f:
            return json.load(f)
    return []


def _save_news(data: list):
    with open(NEWS_FILE, "w") as f:
        json.dump(data, f, indent=4)


def _news_id() -> str:
    return "".join(random.choices(string.ascii_lowercase + string.digits, k=10))


class AddNewsRequest(BaseModel):
    AdminKey: str
    tag: str            # NEW | UPDATE | EVENT
    title: str
    body: str
    date: str = ""      # e.g. "May 3, 2025" — blank = today
    image_url: str = ""
    published: bool = True


class EditNewsRequest(BaseModel):
    AdminKey: str
    id: str
    tag: str
    title: str
    body: str
    date: str = ""
    image_url: str = ""
    published: bool = True


@app.post("/admin/news/add")
def admin_news_add(req: AddNewsRequest):
    if req.AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    if not req.title or not req.body:
        return {"log": "MissingFields"}
    news = _load_news()
    today = datetime.now()
    day = str(today.day)
    date_str = req.date or f"{today.strftime('%B')} {day}, {today.year}"
    post = {
        "id":         _news_id(),
        "tag":        req.tag.upper(),
        "title":      req.title,
        "body":       req.body,
        "date":       date_str,
        "image_url":  req.image_url,
        "published":  req.published,
        "created_at": today.strftime("%Y-%m-%d %H:%M:%S"),
    }
    news.insert(0, post)  # newest first
    _save_news(news)
    print(f"[News] Added post: {post['title']}")
    return {"log": "ok", "post": post}


@app.post("/admin/news/edit")
def admin_news_edit(req: EditNewsRequest):
    if req.AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    news = _load_news()
    for i, post in enumerate(news):
        if post["id"] == req.id:
            news[i].update({
                "tag":       req.tag.upper(),
                "title":     req.title,
                "body":      req.body,
                "date":      req.date or post.get("date", ""),
                "image_url": req.image_url,
                "published": req.published,
            })
            _save_news(news)
            return {"log": "ok", "post": news[i]}
    return {"log": "PostNotFound"}


@app.get("/admin/news/get")
def admin_news_get(AdminKey: str):
    if AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    return {"log": "ok", "news": _load_news()}


@app.get("/admin/news/delete")
def admin_news_delete(AdminKey: str, id: str):
    if AdminKey != ADMIN_KEY:
        return {"log": "InvalidAdminKey"}
    news = _load_news()
    before = len(news)
    news = [p for p in news if p["id"] != id]
    if len(news) == before:
        return {"log": "PostNotFound"}
    _save_news(news)
    return {"log": "ok"}


@app.get("/news")
def public_news():
    """Public endpoint — launcher fetches published posts from here."""
    news = _load_news()
    published = [p for p in news if p.get("published", True)]
    return {"log": "ok", "news": published}
