@app.post("/reports/submit")
def appsubmitreport(request: ReportRequest):
    JFile = getuserfilebytoken(request.Token)
    if JFile == "MissingToken":
        return {"log": "TokenDoesNotExist"}
    
    reporter = JFile["AccountName"]
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    report_data = JFile.get("ReportData", {})
    last_report_time = report_data.get("LastReportTime", "")
    report_count = report_data.get("ReportCount", 0)
    
    if last_report_time:
        try:
            last_time = datetime.strptime(last_report_time, "%Y-%m-%d %H:%M:%S")
            time_diff = (datetime.now() - last_time).total_seconds()
            if time_diff < 300:
                return {"log": "ReportLimitReached"}
        except:
            pass
    
    report_id = ''.join(random.choice(string.ascii_uppercase + string.digits) for _ in range(16))
    
    report_record = {
        "ReportID": report_id,
        "Reporter": reporter,
        "ReportedPlayer": request.ReportedPlayer,
        "Reason": request.Reason,
        "Description": request.Description,
        "Timestamp": current_time,
        "Status": "Pending"
    }
    
    with open(get_path("Reports", report_id + ".json"), "w") as reportfile:
        json.dump(report_record, reportfile, indent=4)
    
    report_data["LastReportTime"] = current_time
    report_data["ReportCount"] = report_count + 1
    JFile["ReportData"] = report_data
    
    with open(get_path("Accounts", reporter + ".json"), "w") as savefile:
        json.dump(JFile, savefile, indent=4)
    
    return {"log": "ok", "reportId": report_id}

def readuserfile(ID: str):
    """Read user account file"""
    try:
        with open(get_path("Accounts", ID + ".json"), "r") as savefile:
            return json.load(savefile)
    except FileNotFoundError:
        return None
    except Exception as e:
        print(f"Error reading user file for {ID}: {e}")
        return None

def getuserfilebytoken(Token: str):
    """Get user file by token"""
    for player_id, token in TokenDict.items():
        if token == Token:
            return readuserfile(player_id)
    return "MissingToken"

def addtoken(ID: str):
    """Generate and store new token"""
    Token = ''.join(random.choice(string.ascii_uppercase + string.digits) for _ in range(30))
    TokenDict[ID] = Token
    return Token

def removetoken(ID: str):
    """Remove token from active tokens"""
    if ID in TokenDict:
        del TokenDict[ID]

def createaccount(ID: str, Pass: str):
    """Create new account"""
    if len(ID) < 3:
        return {"log": "IDTooShort"}
    if len(Pass) < 9:
        return {"log": "PassTooShort"}
    
    if os.path.exists(get_path("Accounts", ID + ".json")):
        return {"log": "ExistingID"}
    
    try:
        hashed = bcrypt.hashpw(Pass.encode('utf-8'), bcrypt.gensalt())
    except Exception as e:
        print(f"Bcrypt error: {e}")
        return {"log": "ServerError"}
    
    account_data = {
        "AccountName": ID,
        "Pass": hashed.decode('utf-8'),
        "Level": 1,
        "XP": 0,
        "CloudCoins": 0,
        "OwnedSkins": [],
        "EquippedSkin": "DefaultSkin",
        "BanStatus": {
            "IsBanned": False,
            "BanReason": "",
            "BannedAt": "",
            "BannedBy": "",
            "ExpiresAt": None
        },
        "ReportData": {
            "LastReportTime": "",
            "ReportCount": 0
        }
    }
    
    try:
        with open(get_path("Accounts", ID + ".json"), "w") as savefile:
            json.dump(account_data, savefile, indent=4)
    except Exception as e:
        print(f"Error creating account file: {e}")
        return {"log": "ServerError"}
    
    print(f"Account created: {ID}")
    return {"log": "AccountCreated", "Token": addtoken(ID)}

def auth(ID: str, Pass: str):
    """Authenticate user"""
    if not os.path.exists(get_path("Accounts", ID + ".json")):
        return {"log": "UserNameNotExist"}
    
    ban_status = isbanned(ID)
    if ban_status["banned"]:
        return {
            "log": "AccountBanned",
            "reason": ban_status["reason"],
            "expiresAt": ban_status.get("expiresAt")
        }
    
    JFile = readuserfile(ID)
    if not JFile:
        return {"log": "UserNameNotExist"}
    
    try:
        AccPass = JFile["Pass"]
        if bcrypt.checkpw(Pass.encode('utf-8'), AccPass.encode('utf-8')):
            return {"log": "ConnectionOK", "token": addtoken(ID)}
        else:
            return {"log": "PassWrong"}
    except Exception as e:
        print(f"Auth error for {ID}: {e}")
        return {"log": "ServerError"}

def isbanned(ID: str):
    """Check if player is banned"""
    JFile = readuserfile(ID)
    if not JFile:
        return {"banned": False}
    
    ban_status = JFile.get("BanStatus", {})
    is_banned = ban_status.get("IsBanned", False)
    
    if not is_banned:
        return {"banned": False}
    
    expires_at = ban_status.get("ExpiresAt")
    if expires_at:
        try:
            expire_time = datetime.strptime(expires_at, "%Y-%m-%d %H:%M:%S")
            if datetime.now() >= expire_time:
                JFile["BanStatus"]["IsBanned"] = False
                with open(get_path("Accounts", ID + ".json"), "w") as savefile:
                    json.dump(JFile, savefile, indent=4)
                return {"banned": False}
        except:
            pass
    
    return {
        "banned": True,
        "reason": ban_status.get("BanReason", "No reason provided"),
        "bannedAt": ban_status.get("BannedAt", ""),
        "expiresAt": expires_at
    }

def banplayer(ID: str, Reason: str, BannedBy: str, ExpiresAt: str = None):
    """Ban a player"""
    if not os.path.exists(get_path("Accounts", ID + ".json")):
        return {"log": "PlayerNotFound"}
    
    JFile = readuserfile(ID)
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    JFile["BanStatus"] = {
        "IsBanned": True,
        "BanReason": Reason,
        "BannedAt": current_time,
        "BannedBy": BannedBy,
        "ExpiresAt": ExpiresAt
    }
    
    with open(get_path("Accounts", ID + ".json"), "w") as savefile:
        json.dump(JFile, savefile, indent=4)
    
    ban_record = {
        "PlayerID": ID,
        "Reason": Reason,
        "BannedBy": BannedBy,
        "BannedAt": current_time,
        "ExpiresAt": ExpiresAt,
        "Active": True
    }
    
    with open(get_path("Bans", ID + ".json"), "w") as banfile:
        json.dump(ban_record, banfile, indent=4)
    
    removetoken(ID)
    print(f"Player banned: {ID}")
    return {"log": "PlayerBanned"}

def unbanplayer(ID: str):
    """Unban a player"""
    if not os.path.exists(get_path("Accounts", ID + ".json")):
        return {"log": "PlayerNotFound"}
    
    JFile = readuserfile(ID)
    JFile["BanStatus"]["IsBanned"] = False
    
    with open(get_path("Accounts", ID + ".json"), "w") as savefile:
        json.dump(JFile, savefile, indent=4)
    
    ban_file = get_path("Bans", ID + ".json")
    if os.path.exists(ban_file):
        os.remove(ban_file)
    
    print(f"Player unbanned: {ID}")
    return {"log": "PlayerUnbanned"}

def getactivebans():
    """Get list of all active bans"""
    bans = []
    ban_dir = get_path("Bans")
    
    if not os.path.exists(ban_dir):
        return bans
    
    try:
        for filename in os.listdir(ban_dir):
            if filename.endswith(".json"):
                with open(get_path("Bans", filename), "r") as banfile:
                    ban_data = json.load(banfile)
                    bans.append(ban_data)
    except Exception as e:
        print(f"Error reading bans: {e}")
    
    return bans
